cmake_minimum_required(VERSION 3.20)
project(argo_camera_view LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GBM REQUIRED IMPORTED_TARGET gbm egl vulkan)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-app-1.0 gstreamer-video-1.0)
set(IHS_PREFIX "$ENV{IHS_PREFIX}" CACHE PATH "Matched IHS prefix")
find_path(IHS_INCLUDE_DIR ihs/platform_view.h PATHS "${IHS_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
find_library(IHS_SHARED_LIBRARY ihs_shared PATHS "${IHS_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
find_path(RAPIDJSON_INCLUDE_DIR rapidjson/document.h
  HINTS "$ENV{RAPIDJSON_ROOT}/include" "${IHS_PREFIX}/include"
        "${CMAKE_CURRENT_SOURCE_DIR}/../../../../ivi-homescreen/third_party/rapidjson/include" REQUIRED)
add_library(argo_camera_view SHARED src/camera_view.cpp)
target_include_directories(argo_camera_view PRIVATE "${IHS_INCLUDE_DIR}")
target_include_directories(argo_camera_view SYSTEM PRIVATE "${RAPIDJSON_INCLUDE_DIR}")
target_compile_features(argo_camera_view PRIVATE cxx_std_20)
target_compile_options(argo_camera_view PRIVATE -Wall -Wextra -Werror)
target_link_libraries(argo_camera_view PRIVATE "${IHS_SHARED_LIBRARY}" PkgConfig::GBM PkgConfig::GST)
set_target_properties(argo_camera_view PROPERTIES CXX_VISIBILITY_PRESET hidden)
include(CTest)
if(BUILD_TESTING)
 add_executable(camera_contract_test test/contract_test.cpp)
 target_include_directories(camera_contract_test SYSTEM PRIVATE "${RAPIDJSON_INCLUDE_DIR}")
 target_compile_features(camera_contract_test PRIVATE cxx_std_20)
 target_compile_options(camera_contract_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
 add_test(NAME camera_contract COMMAND camera_contract_test)
 add_executable(camera_static_image_state_test test/static_image_state_test.cpp)
 target_compile_features(camera_static_image_state_test PRIVATE cxx_std_20)
 target_compile_options(camera_static_image_state_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
 add_test(NAME camera_static_image_state COMMAND camera_static_image_state_test)
endif()
